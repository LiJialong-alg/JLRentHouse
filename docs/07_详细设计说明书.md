```java
package org.example.lease.common.result;

/**
 * 统一响应结果

@Data
public class Result<T> implements Serializable {
    private static final long serialVersionUID = 1L;

    private Integer code;      // 状态码
    private String message;    // 消息
    private T data;            // 数据

    public static <T> Result<T> ok() {
        return ok(null);
    }

    public static <T> Result<T> ok(T data) {
        Result<T> result = new Result<>();
        result.setCode(ResultCodeEnum.SUCCESS.getCode());
        result.setMessage(ResultCodeEnum.SUCCESS.getMessage());
        result.setData(data);
        return result;
    }

    public static <T> Result<T> fail(String message) {
        Result<T> result = new Result<>();
        result.setCode(ResultCodeEnum.ERROR.getCode());
        result.setMessage(message);
        return result;
    }

    public static <T> Result<T> fail(Integer code, String message) {
        Result<T> result = new Result<>();
        result.setCode(code);
        result.setMessage(message);
        return result;
    }
}
```

#### 2.1.3 JWT工具类 (JWTUtil)

```java
package org.example.lease.common.utils;

@Component
public class JWTUtil {
    private static final String SECRET_KEY = "lease_secret_key_2026";
    private static final long EXPIRE_TIME = 24 * 60 * 60 * 1000; // 24小时

    /**
     * 生成Token
     */
    public String createToken(Long userId, String username) {
        return Jwts.builder()
                .setSubject(username)
                .claim("userId", userId)
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + EXPIRE_TIME))
                .signWith(SignatureAlgorithm.HS512, SECRET_KEY)
                .compact();
    }

    /**
     * 解析Token
     */
    public Claims parseToken(String token) {
        return Jwts.parser()
                .setSigningKey(SECRET_KEY)
                .parseClaimsJws(token)
                .getBody();
    }

    /**
     * 验证Token
     */
    public boolean validateToken(String token) {
        try {
            parseToken(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    /**
     * 获取用户ID
     */
    public Long getUserId(String token) {
        Claims claims = parseToken(token);
        return claims.get("userId", Long.class);
    }
}
```

```java
@Service
public class LoginServiceImpl implements LoginService {

    @Autowired
    private SystemUserService systemUserService;
    @Autowired
    private JWTUtil jwtUtil;
    @Autowired
    private StringRedisTemplate redisTemplate;

    @Override
    public LoginVo login(LoginDto loginDto) {
        // 1. 查询用户
        SystemUser systemUser = systemUserService.lambdaQuery()
                .eq(SystemUser::getUsername, loginDto.getUsername())
                .one();

        if (systemUser == null) {
            throw new LeaseException(ResultCodeEnum.USER_NOT_EXIST_601);
        }

        // 2. 验证密码
        if (!PasswordEncoder.matches(loginDto.getPassword(), systemUser.getPassword())) {
            throw new LeaseException(ResultCodeEnum.PASSWORD_ERROR_602);
        }

        // 3. 检查状态
        if (systemUser.getStatus() == SystemUserStatus.DISABLED) {
            throw new LeaseException("账号已被禁用");
        }

        // 4. 生成Token
        String token = jwtUtil.createToken(systemUser.getId(), systemUser.getUsername());

        // 5. 缓存用户信息到Redis
        redisTemplate.opsForValue().set(
                RedisConstant.ADMIN_TOKEN_PREFIX + token,
                JSON.toJSONString(systemUser),
                RedisConstant.ADMIN_TOKEN_EXPIRE,
                TimeUnit.SECONDS
        );

        // 6. 构建返回结果
        LoginVo loginVo = new LoginVo();
        loginVo.setToken(token);
        loginVo.setUserInfo(systemUser);
        return loginVo;
    }

    @Override
    public void logout() {
        String token = LoginUserHolder.getToken();
        if (StringUtils.hasText(token)) {
            redisTemplate.delete(RedisConstant.ADMIN_TOKEN_PREFIX + token);
        }
    }
}
```

```java
@Service
public class ApartmentServiceImpl extends IServiceImpl<ApartmentInfoMapper, ApartmentInfo>
        implements ApartmentService {

    @Autowired
    private ApartmentInfoMapper apartmentInfoMapper;
    @Autowired
    private GraphInfoService graphInfoService;

    @Override
    public IPage<ApartmentVo> pageApartment(Page<ApartmentVo> page, ApartmentQueryVo queryVo) {
        return apartmentInfoMapper.selectPageApartment(page, queryVo);
    }

    @Override
    public ApartmentVo getApartmentInfoById(Long id) {
        // 1. 获取公寓基本信息
        ApartmentInfo apartmentInfo = baseMapper.selectById(id);
        if (apartmentInfo == null) {
            throw new LeaseException("公寓不存在");
        }

        // 2. 转换为VO
        ApartmentVo apartmentVo = new ApartmentVo();
        BeanUtils.copyProperties(apartmentInfo, apartmentVo);

        // 3. 获取配套列表
        List<FacilityInfo> facilities = graphInfoService.getFacilitiesByApartmentId(id);
        apartmentVo.setFacilityList(facilities);

        // 4. 获取标签列表
        List<LabelInfo> labels = graphInfoService.getLabelsByApartmentId(id);
        apartmentVo.setLabelList(labels);

        // 5. 获取图片列表
        List<GraphInfo> images = graphInfoService.getImagesByApartmentId(id);
        apartmentVo.setGraphInfoList(images);

        return apartmentVo;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void saveApartment(ApartmentFormVo formVo) {
        // 1. 保存公寓基本信息
        ApartmentInfo apartmentInfo = new ApartmentInfo();
        BeanUtils.copyProperties(formVo, apartmentInfo);
        baseMapper.insert(apartmentInfo);

        Long apartmentId = apartmentInfo.getId();

        // 2. 保存配套设施关系
        if (CollectionUtils.isNotEmpty(formVo.getFacilityIds())) {
            saveFacilityRelations(apartmentId, formVo.getFacilityIds());
        }

        // 3. 保存标签关系
        if (CollectionUtils.isNotEmpty(formVo.getLabelIds())) {
            saveLabelRelations(apartmentId, formVo.getLabelIds());
        }

        // 4. 保存图片
        if (CollectionUtils.isNotEmpty(formVo.getGraphInfoList())) {
            saveImageRelations(GraphInfoType.APARTMENT, apartmentId, formVo.getGraphInfoList());
        }
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void updateApartment(ApartmentFormVo formVo) {
        // 1. 更新公寓基本信息
        ApartmentInfo apartmentInfo = new ApartmentInfo();
        BeanUtils.copyProperties(formVo, apartmentInfo);
        baseMapper.updateById(apartmentInfo);

        Long apartmentId = formVo.getId();

        // 2. 更新配套设施关系
        graphInfoService.deleteFacilityRelationsByApartmentId(apartmentId);
        if (CollectionUtils.isNotEmpty(formVo.getFacilityIds())) {
            saveFacilityRelations(apartmentId, formVo.getFacilityIds());
        }

        // 3. 更新标签关系
        graphInfoService.deleteLabelRelationsByApartmentId(apartmentId);
        if (CollectionUtils.isNotEmpty(formVo.getLabelIds())) {
            saveLabelRelations(apartmentId, formVo.getLabelIds());
        }

        // 4. 更新图片
        graphInfoService.deleteImageRelations(GraphInfoType.APARTMENT, apartmentId);
        if (CollectionUtils.isNotEmpty(formVo.getGraphInfoList())) {
            saveImageRelations(GraphInfoType.APARTMENT, apartmentId, formVo.getGraphInfoList());
        }
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void removeApartmentById(Long id) {
        // 1. 删除公寓基本信息
        baseMapper.deleteById(id);

        // 2. 删除关联数据
        graphInfoService.deleteAllRelationsByApartmentId(id);
    }
}
```

```java
@Service
public class FileUploadServiceImpl implements FileUploadService {

    @Autowired
    private MinioProperties minioProperties;

    @Override
    public String uploadImage(MultipartFile file) {
        try {
            // 1. 验证文件
            validateFile(file);

            // 2. 生成文件名
            String filename = generateFilename(file.getOriginalFilename());

            // 3. 上传到MinIO
            PutObjectArgs args = PutObjectArgs.builder()
                    .stream(file.getInputStream(), file.getSize(), -1)
                    .bucket(minioProperties.getBucketName())
                    .object(filename)
                    .contentType(file.getContentType())
                    .build();
            minioProperties.getClient().putObject(args);

            // 4. 返回URL
            return minioProperties.getExternalUrl() + "/" + minioProperties.getBucketName() + "/" + filename;
        } catch (Exception e) {
            throw new LeaseException("文件上传失败: " + e.getMessage());
        }
    }

    private void validateFile(MultipartFile file) {
        if (file.isEmpty()) {
            throw new LeaseException("文件不能为空");
        }

        // 文件大小限制 10MB
        if (file.getSize() > 10 * 1024 * 1024) {
            throw new LeaseException("文件大小不能超过10MB");
        }

        // 文件类型限制
        String contentType = file.getContentType();
        if (contentType == null || !contentType.startsWith("image/")) {
            throw new LeaseException("只能上传图片文件");
        }
    }

    private String generateFilename(String originalFilename) {
        String extension = originalFilename.substring(originalFilename.lastIndexOf("."));
        return "images/" + DateFormatUtils.format(new Date(), "yyyyMMdd") + "/" +
               UUID.randomUUID().toString().replace("-", "") + extension;
    }
}
```

#### 3.1.1 axios封装 (utils/http/index.ts)

```typescript
import axios from 'axios'
import { ElMessage } from 'element-plus'
import { useUserStore } from '@/store/modules/user'
import { ResultEnum } from '@/enums/httpEnums'
import router from '@/router'

export const service = axios.create({
  baseURL: import.meta.env.PROD ? import.meta.env.VITE_APP_BASE_URL : '/',
  timeout: ResultEnum.TIMEOUT as number,
})

// 请求拦截器
service.interceptors.request.use(
  (config) => {
    const userStore = useUserStore()
    const token = userStore.token
    if (token) {
      config.headers['access-token'] = token
    }
    return config
  },
  (error: AxiosError) => {
    ElMessage.error(error.message)
    return Promise.reject(error)
  },
)

// 响应拦截器
service.interceptors.response.use(
  (response: AxiosResponse) => {
    const { data } = response
    if (ResultEnum.EXPIRE.includes(data.code)) {
      RESEETSTORE()
      ElMessage.error(data.message || ResultEnum.ERRMESSAGE)
      router.replace(LOGIN_URL)
      return Promise.reject(data)
    }

    if (data.code && data.code !== ResultEnum.SUCCESS) {
      ElMessage.error(data.message || ResultEnum.ERRMESSAGE)
      return Promise.reject(data)
    }
    return data
  },
  (error: AxiosError) => {
    let message = ''
    const status = error.response?.status
    switch (status) {
      case 401:
        message = 'token 失效，请重新登录'
        RESEETSTORE()
        router.replace(LOGIN_URL)
        break
      case 403:
        message = '拒绝访问'
        break
      case 404:
        message = '请求地址错误'
        break
      case 500:
        message = '服务器故障'
        break
      default:
        message = '网络连接故障'
    }
    ElMessage.error(message)
    return Promise.reject(error)
  },
)
```

#### 3.2.1 用户Store (store/modules/user.ts)

```typescript
import { defineStore } from 'pinia'
import { reactive, ref } from 'vue'

export const useUserStore = defineStore('user', () => {
  const token = ref<string>(localStorage.getItem('app-token') || '')
  const userInfo = reactive<any>({})

  const setToken = (newToken: string) => {
    token.value = newToken
    localStorage.setItem('app-token', newToken)
  }

  const setUserInfo = (info: any) => {
    Object.assign(userInfo, info)
    localStorage.setItem('app-user', JSON.stringify(info))
  }

  const RESETSTORE = () => {
    token.value = ''
    localStorage.removeItem('app-token')
    localStorage.removeItem('app-user')
    Object.keys(userInfo).forEach(key => delete userInfo[key])
  }

  return {
    token,
    userInfo,
    setToken,
    setUserInfo,
    RESETSTORE
  }
})
```

## 5. 异常处理设计

### 5.1 自定义异常

```java
@Getter
public class LeaseException extends RuntimeException {
    private final Integer code;
    private final String message;

    public LeaseException(String message) {
        super(message);
        this.code = ResultCodeEnum.ERROR.getCode();
        this.message = message;
    }

    public LeaseException(Integer code, String message) {
        super(message);
        this.code = code;
        this.message = message;
    }

    public LeaseException(ResultCodeEnum resultCodeEnum) {
        super(resultCodeEnum.getMessage());
        this.code = resultCodeEnum.getCode();
        this.message = resultCodeEnum.getMessage();
    }
}
```
